// Long/Short Ratio hesaplama fonksiyonu
function calculateLongShortRatio(longPositions, shortPositions) {
    // Long ve short pozisyonlarının toplamını al
    const totalLong = longPositions.reduce((sum, pos) => sum + pos, 0);
    const totalShort = shortPositions.reduce((sum, pos) => sum + pos, 0);
    
    // Oranı hesapla (short pozisyonlar 0 ise 1 döndür)
    const ratio = totalShort > 0 ? totalLong / totalShort : 1;
    
    return {
        ratio: ratio,
        totalLong: totalLong,
        totalShort: totalShort
    };
}

// Long/Short Ratio görüntüleme fonksiyonu
function displayLongShortRatio(ratioData) {
    const ratioCell = document.createElement('td');
    
    // Oranı formatla ve göster
    ratioCell.textContent = ratioData.ratio.toFixed(2);
    
    // Renklendirme
    if (ratioData.ratio > 1.5) {
        ratioCell.className = 'positive'; // Çok fazla long pozisyon
    } else if (ratioData.ratio > 1) {
        ratioCell.className = 'weak-positive'; // Hafif fazla long pozisyon
    } else if (ratioData.ratio < 0.5) {
        ratioCell.className = 'negative'; // Çok fazla short pozisyon
    } else if (ratioData.ratio < 1) {
        ratioCell.className = 'weak-negative'; // Hafif fazla short pozisyon
    } else {
        ratioCell.className = 'neutral'; // Dengeli
    }
    
    // Tooltip ekle (üzerine gelindiğinde detayları göster)
    ratioCell.title = `Long: ${ratioData.totalLong.toLocaleString()}\nShort: ${ratioData.totalShort.toLocaleString()}`;
    
    return ratioCell;
}

// Long/Short Ratio yorumlama mantığı:
/*
1. Oran > 1.5: Güçlü Long Baskısı
   - Çok fazla long pozisyon var
   - Fiyat yükseliş eğiliminde olabilir
   - Dikkat: Aşırı alım sinyali olabilir

2. Oran > 1: Hafif Long Baskısı
   - Long pozisyonlar biraz daha fazla
   - Yükseliş eğilimi olabilir
   - Normal piyasa koşulları

3. Oran = 1: Dengeli
   - Long ve short pozisyonlar eşit
   - Piyasa dengede
   - Yön belirsiz

4. Oran < 1: Hafif Short Baskısı
   - Short pozisyonlar biraz daha fazla
   - Düşüş eğilimi olabilir
   - Normal piyasa koşulları

5. Oran < 0.5: Güçlü Short Baskısı
   - Çok fazla short pozisyon var
   - Fiyat düşüş eğiliminde olabilir
   - Dikkat: Aşırı satım sinyali olabilir
*/
