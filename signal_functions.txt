// Signal hesaplama fonksiyonu
function generateSignal(rsi, macdData, priceChange) {
    // RSI sinyalleri - daha hassas bölgeler
    let rsiSignal = 'NEUTRAL';
    if (rsi > 75) rsiSignal = 'SHORT';
    else if (rsi > 65) rsiSignal = 'WEAK_SHORT';
    else if (rsi < 25) rsiSignal = 'LONG';
    else if (rsi < 35) rsiSignal = 'WEAK_LONG';
    
    // MACD sinyalleri - histogram değerine göre
    let macdSignal = 'NEUTRAL';
    const macdStrength = Math.abs(macdData.histogram);
    if (macdData.histogram > 0) {
        macdSignal = macdStrength > 10 ? 'LONG' : 'WEAK_LONG';
    } else if (macdData.histogram < 0) {
        macdSignal = macdStrength > 10 ? 'SHORT' : 'WEAK_SHORT';
    }
    
    // Fiyat değişimi sinyalleri - yüzdeye göre
    let priceSignal = 'NEUTRAL';
    if (priceChange > 2) priceSignal = 'LONG';
    else if (priceChange > 0.5) priceSignal = 'WEAK_LONG';
    else if (priceChange < -2) priceSignal = 'SHORT';
    else if (priceChange < -0.5) priceSignal = 'WEAK_SHORT';

    // Sinyal gücü hesaplama
    let signalStrength = 0;
    if (rsiSignal === 'LONG' || macdSignal === 'LONG' || priceSignal === 'LONG') signalStrength += 2;
    if (rsiSignal === 'WEAK_LONG' || macdSignal === 'WEAK_LONG' || priceSignal === 'WEAK_LONG') signalStrength += 1;
    if (rsiSignal === 'SHORT' || macdSignal === 'SHORT' || priceSignal === 'SHORT') signalStrength -= 2;
    if (rsiSignal === 'WEAK_SHORT' || macdSignal === 'WEAK_SHORT' || priceSignal === 'WEAK_SHORT') signalStrength -= 1;

    // Final sinyal belirleme
    if (signalStrength >= 3) return 'LONG';
    if (signalStrength >= 1) return 'WEAK_LONG';
    if (signalStrength <= -3) return 'SHORT';
    if (signalStrength <= -1) return 'WEAK_SHORT';
    return 'NEUTRAL';
}

// Signal görüntüleme fonksiyonu
function displaySignal(signal) {
    const signalCell = document.createElement('td');
    signalCell.textContent = signal;
    
    // Signal renklendirme
    if (signal === 'LONG') {
        signalCell.className = 'long';
    } else if (signal === 'WEAK_LONG') {
        signalCell.className = 'weak-long';
    } else if (signal === 'SHORT') {
        signalCell.className = 'short';
    } else if (signal === 'WEAK_SHORT') {
        signalCell.className = 'weak-short';
    } else {
        signalCell.className = 'neutral';
    }
    
    return signalCell;
}
